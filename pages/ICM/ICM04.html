Raining Day

Create random raindrops using a loop. 


    function setup() {
        createCanvas(500, 300);
        noLoop();
    }

    function draw() {
        background(255);
        noFill();

        let rainX1;
        let rainY1;
        let rainX2;
        let rainY2;

        for(let i=1; i<=12; i++){
            rainX1 = random( 30 + (i-1) * 35 , 50 + 35 * i );
            rainY1 = random( 10 , 100);
            rainX2 = rainX1 + random( 10 , 40);
            rainY2 = random( 130 , 270 );

            let widths = [20, 30, 50, 80, 100];
            let heights = [5, 10, 20, 30, 40];

            let rainLength = dist(rainX1, rainY1, rainX2, rainY2);
            let n;
    
            let R = random( 0 , 50);
            let G = map(rainLength, 0, 250, 200, 0);
            let B = map(rainLength, 0, 250, 255, 200);

            for( let I = 0; I < rainLength; I++){
                stroke( R, G, B, 255-I);
            }
        
            line(rainX1, rainY1, rainX2, rainY2);
        
            if ( rainLength > 160 ) {
                n = 5;
            } else if ( rainLength > 100 ) {
                n = 3;
            } else {
                n = 2;
            }

            for (let j = 0; j < n; j++) {
                let w = widths[j];
                let h = heights[j];
                let start = random( 0, TWO_PI );
                let stop =  random( 0, TWO_PI );

                stroke( R , G , B , 255 - j*50 );
                arc( rainX2, rainY2, w, h, start, stop );
            }
        }
    }

Multiple Raindrops and Ripples

Use nested for loops to generate multiple raindrops and create ripples with random shapes and sizes. 



    let rainX1;

    let rainY2;

    for(let i=1; i<=13; i++){
        stroke(random(0,200), random(100,235) , 255);

        line(rainX1, rainY1, rainX2, rainY2);

        let widths = [20, 30, 50, 80, 100];
