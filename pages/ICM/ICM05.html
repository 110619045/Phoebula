加上背景

用push amd pop畫上背景速度線 


function setup() {
  createCanvas(400, 400);
  frameRate(6);
}

function draw() {
  background(0);
  
  for(let i=5;i < 400; i += 6){
      for( let j=5;j < 400;j+=6){
      strokeWeight(0);
      let r = random(200);
      let g = random(200);
      let b = 200;
          fill(r, g, b);
          circle(i,j,5);
      }
  }

  let X = constrain(mouseX, 55, 350);
  let Y = constrain(mouseY, 85, 360);
  
  push();
  fill(255,255,255,200);
  let lazerX = 200;
  let lazerY = 200;
  let n = 20;
  translate(lazerX,lazerX);

  for(let i = 0; i < n; i++){
    rotate(TWO_PI / n);
    triangle(0, 0, -30, -lazerX-100, 30, -lazerY-100);
  }

  pop();
  
  rabbit(X,Y);
}
