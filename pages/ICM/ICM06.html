    function setup() {
        createCanvas(400, 400);
        noLoop();
    }

    function draw() {
        background(220);

        let x = 200;
        let y = 200;
        let r = 30;
        let n = 18;

        stroke(255);
        for(let i=0; i < n ; i++) {
            line(x, y, x + r * cos(TWO_PI / n * i), y + r * sin(TWO_PI / n * i));
            console.log(i);
        }
    }

陣列儲存

用陣列存資料，用push儲存新資料 


    let x = [];
    let y = [];
    let r = [];
    let n = [];
    let hasClicked = false;
    ...

    function draw() {
        background(220);

        if (hasClicked) {
            stroke(255);
            for(let j=0; j < x.length; j++){
                for(let i=0; i < n[j] ; i++){
                    line(x[j], y[j], x[j] + r[j] * cos(TWO_PI / n[j] * i), y[j] + r[j] * sin(TWO_PI / n[j] * i));
                    console.log(i);
                }
            }
        }
    }

    function mousePressed(){
        hasClicked = true;
        x.push(mouseX);
        y.push(mouseY);
        r.push(random(20,80));
        n.push(floor(random(12, 21)));
    }
